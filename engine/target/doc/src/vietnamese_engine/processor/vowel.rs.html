1//! Reading vowel qualities from canonical ASCII raw text.

3use super::rules::{
4    horn_of, MAX_ATOMS, Q_A, Q_ABREVE, Q_ACIRC, Q_E, Q_ECIRC, Q_I, Q_NONE, Q_O, Q_OCIRC, Q_OHORN,
5    Q_U, Q_UHORN, Q_Y,
6};
7
8pub(crate) struct Run {
9    pub(super) atoms: [u8; MAX_ATOMS],
10    pub(super) count: usize,
11    pub(super) end: usize,
12}

14/// Greedily reads vowel qualities starting at `start`.
15///
16/// Encodings accepted per letter:
17///
18/// - doubled letters (`aa`, `ee`, `oo`) carry circumflex;
19/// - `a` followed by `w` carries breve;
20/// - one `w` horns every immediately preceding run of plain `o`/`u`
21///   letters (`uow` → ươ), and an unconsumed `w` is itself the quality ư.
22pub(crate) fn read_run(bytes: &[u8], start: usize) -> Option<Run> {
23    let mut atoms = [Q_NONE; MAX_ATOMS];
24    let mut count = 0usize;
25    let mut horn_from = 0usize;
26    let mut i = start;
27    macro_rules! push {
28        ($quality:expr) => {{
29            if count == MAX_ATOMS {
30                return None;
31            }
32            atoms[count] = $quality;
33            count += 1;
34        }};
35    }
36    while i < bytes.len() {
37        match bytes[i] {
38            b'a' => match bytes.get(i + 1) {
39                Some(b'a') => {
40                    push!(Q_ACIRC);
41                    horn_from = count;
42                    i += 2;
43                }
44                Some(b'w') => {
45                    push!(Q_ABREVE);
46                    horn_from = count;
47                    i += 2;
48                }
49                _ => {
50                    push!(Q_A);
51                    horn_from = count;
52                    i += 1;
53                }
54            },
55            b'e' => {
56                if bytes.get(i + 1) == Some(&b'e') {
57                    push!(Q_ECIRC);
58                    i += 2;
59                } else {
60                    push!(Q_E);
61                    i += 1;
62                }
63                horn_from = count;
64            }
65            b'i' | b'y' => {
66                push!(if bytes[i] == b'i' { Q_I } else { Q_Y });
67                horn_from = count;
68                i += 1;
69            }
70            b'o' => {
71                if bytes.get(i + 1) == Some(&b'o') {
72                    push!(Q_OCIRC);
73                    horn_from = count;
74                    i += 2;
75                } else {
76                    push!(Q_O);
77                    i += 1;
78                }
79            }
80            b'u' => {
81                push!(Q_U);
82                i += 1;
83            }
84            b'w' => {
85                if horn_from < count {
86                    for quality in &mut atoms[horn_from..count] {
87                        if let Some(horned) = horn_of(*quality) {
88                            *quality = horned;
89                        }
90                    }
91                } else {
92                    push!(Q_UHORN);
93                }
94                horn_from = count;
95                i += 1;
96            }
97            _ => break,
98        }
99    }
100    Some(Run {
101        atoms,
102        count,
103        end: i,
104    })
105}

107#[inline]
108pub(crate) fn is_circumflex(quality: u8) -> bool {
109    matches!(quality, Q_ACIRC | Q_ECIRC | Q_OCIRC)
110}
111
112#[inline]
113pub(crate) fn is_breve(quality: u8) -> bool {
114    quality == Q_ABREVE
115}
116
117#[inline]
118pub(crate) fn is_horn(quality: u8) -> bool {
119    matches!(quality, Q_OHORN | Q_UHORN)
120}
121
122/// Byte width of one vowel atom inside its canonical spelling.
123pub(crate) fn atom_width(atoms: &[u8; MAX_ATOMS], index: usize) -> usize {
124    let quality = atoms[index];
125    let mut width = 1;
126    if is_circumflex(quality) || is_breve(quality) {
127        width += 1;
128    }
129    if is_horn(quality) && !atoms.get(index + 1).copied().is_some_and(is_horn) {
130        width += 1;
131    }
132    width
133}
