1//! Tone markers and tone placement rules.
2
3use super::rules::{
4    Coda, VowelRule, MAX_ATOMS, Q_ABREVE, Q_ACIRC, Q_ECIRC, Q_OCIRC, Q_OHORN, Q_UHORN,
5};
6use crate::character::Tone;
7
8/// Vietnamese orthography mode.
9#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
10pub enum Orthography {
11    #[default]
12    Modern,
13    Old,
14}
15
16#[inline]
17pub(crate) fn is_marker_byte(byte: u8) -> bool {
18    matches!(byte, b's' | b'f' | b'r' | b'x' | b'j' | b'z')
19}
20
21#[inline]
22pub(crate) fn marker_tone(byte: u8) -> Option<Tone> {
23    match byte {
24        b's' => Some(Tone::Acute),
25        b'f' => Some(Tone::Grave),
26        b'r' => Some(Tone::Hook),
27        b'x' => Some(Tone::Tilde),
28        b'j' => Some(Tone::Dot),
29        _ => None,
30    }
31}
32
33pub(crate) fn tone_marker(tone: Tone) -> Option<u8> {
34    match tone {
35        Tone::None => None,
36        Tone::Acute => Some(b's'),
37        Tone::Grave => Some(b'f'),
38        Tone::Hook => Some(b'r'),
39        Tone::Tilde => Some(b'x'),
40        Tone::Dot => Some(b'j'),
41    }
42}
43
44pub(crate) fn tone_index(tone: Tone) -> usize {
45    tone as usize
46}
47
48/// Consumes a run of tone-marker bytes, last one winning; `z` cancels.
49pub(crate) fn read_tones(bytes: &[u8], slot: &mut usize) -> Option<Tone> {
50    let mut tone = None;
51    while *slot < bytes.len() && is_marker_byte(bytes[*slot]) {
52        tone = marker_tone(bytes[*slot]);
53        *slot += 1;
54    }
55    tone
56}
57
58/// Relocates a tone marker typed after the coda (`vangf` → grave on `a`).
59pub(crate) fn rescue_trailing_tone(
60    bytes: &[u8],
61    coda_start: usize,
62) -> Option<(usize, Option<Tone>)> {
63    let mut end = bytes.len();
64    while end > coda_start && is_marker_byte(bytes[end - 1]) {
65        end -= 1;
66    }
67    if end == bytes.len() || Coda::identify(&bytes[coda_start..end]).is_none() {
68        return None;
69    }
70    let mut slot = end;
71    Some((end, read_tones(bytes, &mut slot)))
72}

74/// Main-vowel atom index for tone placement.
75pub(crate) fn main_index(grammar: &VowelRule, orthography: Orthography, has_coda: bool) -> u8 {
76    if orthography == Orthography::Old {
77        return old_main_index(grammar.count, grammar.atoms, has_coda);
78    }
79    if grammar.main != super::rules::MAIN_NONE {
80        return grammar.main;
81    }
82    priority_main_index(grammar.count, grammar.atoms)
83}
84
85/// Old-orthography heuristic preserved from the previous engine.
86fn old_main_index(count: u8, atoms: [u8; MAX_ATOMS], has_coda: bool) -> u8 {
87    let mut index = usize::from(count == 3 || has_coda);
88    for (candidate, &quality) in atoms.iter().enumerate().take(count as usize) {
89        if matches!(
90            quality,
91            Q_ABREVE | Q_ACIRC | Q_ECIRC | Q_OCIRC | Q_OHORN | Q_UHORN
92        ) {
93            index = candidate;
94        }
95    }
96    index as u8
97}

99/// Quality-priority fallback for sequences without explicit metadata.
100fn priority_main_index(count: u8, atoms: [u8; MAX_ATOMS]) -> u8 {
101    // Ranks per quality: ơ ê ă ô â ư a o e i u y.
102    const RANKS: [u8; 12] = [6, 2, 4, 8, 1, 9, 7, 3, 0, 10, 5, 11];
103    let mut best = (u8::MAX, 0u8);
104    for (index, &quality) in atoms.iter().enumerate().take(count as usize) {
105        let rank = RANKS[quality as usize];
106        if rank < best.0 {
107            best = (rank, index as u8);
108        }
109    }
110    best.1
111}
